\newcommand{\fullpsalm}[1]{%
	% \index[P]{\csname psalm#1Incipit\endcsname}
	\begingroup % Start local group to limit scope
		\clubpenalty=10000 % SET HIGH to protect the Lettrine
		\setlength{\parindent}{.25em}
		\let\psalmtext\empty
		\expandafter\let\expandafter\psalmtext\csname psalm#1Text\endcsname 
		\newif\ifblue
		\bluetrue
		\firstversetrue % Reset toggle for the new psalm
		\newcommand{\processline}[1]{%
			\StrLeft{##1}{1}[\firstletter]%
			\StrGobbleLeft{##1}{1}[\restofline]%
			\par % End the previous verse
			\iffirstverse
				% \penalty10000    % Lock Verse 1 and Verse 2 together
				% \penalty3000    % Separate Verse 1 and 2
				\penalty500    % Separate Verse 1 and 2
				\clubpenalty=150 % REVERT TO 150 for all subsequent verses
				\firstversefalse
			\fi
			\hspace*{.25em}% Start the new verse
			\ifblue
				{\color{Blue}\textbf{\firstletter}}%
				\bluefalse
			\else
				{\color{Red}\textbf{\firstletter}}%
				\bluetrue
			\fi
			\restofline%
		}
		\renewcommand{\\}{\processline}
		\psalmtext
		\par 
	\endgroup
}
